# 反向传播 + 梯度下降更新参数
    def backward(self, X, y_hat, y_true, lr):
        """
        反向传播求梯度 + 更新w和b
        lr：学习率，控制每一步更新幅度
        """
        # 样本数量
        m = X.shape[0]
        
        # 求梯度
        dz = (y_hat - y_true) * y_hat * (1 - y_hat)
        dw = (1 / m) * np.dot(X.T, dz)
        db = (1 / m) * np.sum(dz)
        
        # 梯度下降更新权重和偏置
        self.w = self.w - lr * dw
        self.b = self.b - lr * db
